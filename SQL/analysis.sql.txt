-- 1. Sales by Region
SELECT Region, SUM(Sales) AS total_sales
FROM healthcare_data
GROUP BY Region
ORDER BY total_sales DESC;

-- 2. Sales by Product
SELECT Product, SUM(Sales) AS total_sales
FROM healthcare_data
GROUP BY Product
ORDER BY total_sales DESC;

-- 3. Approval Rate by Region
SELECT
    Region,
    SUM(Claims_Approved) * 100.0 / SUM(Claims_Submitted) AS approval_rate
FROM healthcare_data
GROUP BY Region
ORDER BY approval_rate DESC;

-- 4. Monthly Sales
SELECT
    Month,
    SUM(Sales) AS total_sales
FROM healthcare_data
GROUP BY Month
ORDER BY CASE Month
    WHEN 'Jan' THEN 1
    WHEN 'Feb' THEN 2
    WHEN 'Mar' THEN 3
    WHEN 'Apr' THEN 4
END;

-- 5. Top-performing Region
SELECT Region, SUM(Sales) AS total_sales
FROM healthcare_data
GROUP BY Region
ORDER BY total_sales DESC
LIMIT 1;
